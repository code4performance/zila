Native windows

examples/ui.zila
package main

use ui

func main()
    var win = ui.Window("Customers", 640, 480)

    var name = ui.TextBox()
    var list = ui.ListView()
    var add  = ui.Button("Add")

    add.onClick(func()
        list.add(name.text)
        name.text = ""
    end)

    win.layout(ui.vstack(name, add, list))
    win.show()
    ui.run()
end

The ui library is written in Zila Lang on top of the Win32 API. ui.Window, ui.TextBox, ui.Button and ui.ListView are real native controls; vstack lays them out; the lambda passed to onClick runs when the button is pressed (or Enter is pressed in the text box). Build it with zila examples/ui.zila -o ui.exe.

Async: waiting, files and TCP

examples/async.zila
// async/await: waiting, files and TCP in one program.
//
//   zila examples/async.zila -o async.exe
//
// An `async func` starts running when it is called and returns a Future.
// `await` waits for a future without blocking the other tasks.

package main

use fs
use list
use net

// a plain function may await too: it suspends only the task that calls it
func fetch_title(id: int) -> text
    await sleep(100)
    return "title {id}"
end

async func fetch_title_task(id: int) -> text
    return fetch_title(id)
end

async func echo_server(listener: net.Listener) -> error
    let conn = try await listener.accept()
    let message = try await conn.read(64)
    try await conn.write("echo: {message}")
    await conn.close()
    return
end

func main() -> error
    // three 100 ms waits overlap: all of it takes about 100 ms, not 300
    let titles = await all([fetch_title_task(1), fetch_title_task(2), fetch_title_task(3)])
    print("{titles[0]}, {titles[1]}, {titles[2]}")

    // the same with `map` and a lambda that awaits (one after another here)
    let ids = list.map([1, 2, 3], func(i) await fetch_title_task(i) end)
    print("titles {ids[0].len} {ids[1].len} {ids[2].len}")

    // files
    try await fs.write("zila-example.txt", "written by an async function")
    print(try await fs.read("zila-example.txt"))

    // a TCP server and a client in the same program
    let listener = try await net.listen(0)
    let server = echo_server(listener)
    let client = try await net.connect("127.0.0.1", listener.port)
    try await client.write("hello")
    print(try await client.read(64))
    try await server
    return
end

Three one-hundred-millisecond waits overlap, so await all(...) takes about 100 ms. The program then writes and reads a file and runs a TCP server and client in the same process; try await turns any failure into an early return. See Async.

FFI & low-level

The whole of examples/ffi.zila. See docs/zila-ffi.md for the full FFI and inline-assembly specification.

examples/ffi.zila
package main

use c

// ---- calling C ----

#[link("m")]
extern "C" func sqrt(x: f64) -> f64

extern "C" func printf(fmt: *c.char, ...) -> c.int

extern "C" func qsort(base: *c.void, n: c.size_t, size: c.size_t,
    cmp: extern "C" func(*c.void, *c.void) -> c.int)

#[link_src("vendor/crc.c")]
extern "C" func crc32(data: *u8, len: usize) -> u32

#[stdcall_example]
extern "stdcall" func MessageBoxA(hwnd: usize, text: *c.char, caption: *c.char, kind: u32) -> c.int

#[c_layout]
type Point = record
    x: f64
    y: f64
end

// ---- being called from C ----

#[export("zila_add")]
func add(a: i32, b: i32) -> i32
    return a + b
end

// ---- inline assembly ----

#[cfg("x86_64")]
func rdtsc() -> u64
    var lo: u32 = 0
    var hi: u32 = 0
    unsafe
        asm
            rdtsc
            out eax = lo
            out edx = hi
        end
    end
    return u64(hi) << 32 bitor u64(lo)
end

#[cfg("x86_64")]
func add_asm(a: u64, b: u64) -> u64
    var r: u64 = a
    unsafe
        asm
            add @r, @b
            inout r = r
            in b = b
            clobber cc
        end
    end
    return r
end

#[naked]
#[cfg("x86_64")]
func halt()
    asm
        cli
        hlt
    end
end

func main()
    unsafe
        printf(c.cstr("sqrt(2) = %f\n"), sqrt(2.0))
    end
    print("tsc: {rdtsc()}")
end

This program exercises Zila Lang on both sides of the C boundary. extern "C" func sqrt(...) and printf declare foreign functions pulled in via #[link("m")] (link against libm); printf's trailing ... marks it as C-variadic. qsort's cmp parameter shows an extern "C" func(...) type used as a C-compatible function-pointer callback. #[c_layout] on Point pins field order and alignment to match a C struct exactly, so it can be passed across the FFI boundary safely. #[export("zila_add")] does the reverse — it exposes a Zila Lang function to C callers under a fixed, unmangled symbol name. rdtsc reads the CPU timestamp counter with a short inline asm block, using out eax = lo / out edx = hi to bind two hardware registers to Zila Lang variables. Every one of these operations — calling extern functions and running asm — is wrapped in unsafe, since the compiler can't verify what happens on the other side of the boundary. See the FFI section of the syntax guide and the project's docs/zila-ffi.md for the full specification.

tree: shared objects and weak references

examples/tree.zila
// tree: shared objects with `new`, and weak parent links that do not leak.
//
//   zila examples/tree.zila -o tree.exe
//
// `new Node{ ... }` makes one shared, reference-counted object of type Rc<Node>:
// copying an Rc shares the object. A child that points back at its parent with an
// Rc would make a cycle that is never freed, so it holds a Weak<Node> instead:
// `weak(x)` makes one, `w.upgrade()` returns the parent again while it is alive
// (and nil once the last Rc to it is gone).

package main

type Node = record
    name: text
    parent: Weak<Node>
    children: []Rc<Node>
end

func add_child(parent: Rc<Node>, name: text) -> Rc<Node>
    let child = new Node{ name: name, parent: weak(parent) }
    parent.children.push(child)
    return child
end

func path(node: Rc<Node>) -> text
    let up = node.parent.upgrade()
    if up == nil
        return node.name
    end
    return path(up) + "/" + node.name
end

func build() -> int
    let root = new Node{ name: "root" }
    let src = add_child(root, "src")
    add_child(src, "main.zila")
    add_child(src, "util.zila")
    add_child(root, "docs")
    print(path(src.children[1]))
    return root.children.len
end

func main()
    let before = live_objects()
    print("top-level entries: {build()}")
    print("objects left after the tree is gone: {live_objects() - before}")
end

A tree whose children point back at their parent. new Node{ ... } makes a shared, reference-counted Rc<Node>; the child's parent field is a Weak<Node>, so the parent and its children do not keep each other alive. live_objects() shows that the whole tree is freed when the last outside reference goes away.
