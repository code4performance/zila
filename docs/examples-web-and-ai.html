REST API

examples/rest_api.zila
// A small REST service on `use web`: routes with path parameters, JSON in and out, a middleware.
//
//   zila examples/rest_api.zila -o rest_api.exe
//   rest_api.exe                      (then: curl http://localhost:8080/customers)
//
// The customers live in memory. Handlers are closures that capture by value, so state that
// requests share is a package-level variable. A handler returns `Response or error`; an error
// becomes a status through its kind (`errors.NotFound` is 404, `errors.InvalidInput` is 400).
// `use json` is there because `web.read_json` and `web.json` write code for your types with it.

package main

use errors
use json
use text
use web

type Customer = record
    id: int
    name: text
end

var customers: Map<int, Customer> = {}
var next_id = 1

func list(req: web.Request) -> web.Response or error
    var rows: []Customer = []
    for id, c in customers
        rows.add(c)
    end
    return web.json<[]Customer>(200, rows)
end

func show(req: web.Request) -> web.Response or error
    let id = try text.parse_int(req.param("id") ?? "")
    if not customers.has(id)
        return error(errors.NotFound, "no customer {id}")
    end
    return web.json<Customer>(200, customers[id])
end

func create(req: web.Request) -> web.Response or error
    var c = try web.read_json(Customer, req)
    if c.name == ""
        return error(errors.InvalidInput, "a customer needs a name")
    end
    c.id = next_id
    next_id += 1
    customers[c.id] = c
    return web.created(Customer, c)
end

func main() -> error
    var app = web.server(port: 8080)
    app.use(web.logger())
    app.get("/customers", list)
    app.get("/customers/:id", show)
    app.post("/customers", create)
    print("listening on http://localhost:8080")
    try app.run()
    return
end

A small REST service on use web: web.server(port: 8080) makes the server, app.get, app.post add routes (/customers/:id has a path parameter read with req.param("id")), and app.use(web.logger()) adds a middleware that prints one line per request. A handler returns Response or error: web.json<[]Customer>(200, rows) and web.created(Customer, c) encode with use json (the program writes use json too, because the code that does it is generated for your types), and web.read_json(Customer, req) decodes the body. An error is turned into a status by its kind: errors.NotFound becomes 404 with {"error": "no customer 9"}, errors.InvalidInput becomes 400, and an unknown kind becomes 500 with the fixed body internal error (the message goes to the log, never to the client). The customers are kept in a package-level Map, because handlers are closures that capture by value; there is no database library yet. The web library page lists everything the server, router, client and templates do.

json: read and write your own types

examples/json.zila
// JSON: read and write your own types, or a free-form tree.
//
//   zila examples/json.zila -o json.exe
//
// `json.parse(T, text)` turns JSON into a value of any type you declare, and `json.encode(T, x)`
// does the reverse. There is nothing to register: the library reads the shape of `T` at build
// time (comptime reflection) and writes the code for it. A field attribute renames a key
// (`#[json("host_name")]`) or leaves the field out (`#[json(skip)]`), and a field of type `text?`
// may be `null` or missing. Errors say where: `at $.port` is the path to the bad value, and a
// syntax error says the line and column.

package main

use json

type Config = record
    port: int
    #[json("host_name")] host: text
    proxy: text?
    tags: []text
    #[json(skip)] cache: text
end

func main()
    let raw = "\{\"port\": 8080, \"host_name\": \"example.org\", \"tags\": [\"a\", \"b\"], \"extra\": true\}"
    match json.parse(Config, raw)
        ok(cfg) ->
            print("port: {cfg.port}")
            print("host: {cfg.host}")
            print("proxy: {cfg.proxy ?? "none"}")
            print(json.encode_with(Config, cfg, json.Options { indent: 2 }) ?? "")
        err(e) -> print("bad config: {e.message}")
    end

    match json.parse(Config, "\{\"port\": \"80\"\}")
        ok(cfg) -> print("unexpected")
        err(e) -> print("bad config: {e.message}")
    end

    // a free-form tree, when there is no type to read into
    match json.parse_value("[1, \{\"a\": null\}]")
        ok(v) -> print(json.stringify(v) ?? "")
        err(e) -> print(e.message)
    end
end

json.parse(Config, raw) reads JSON into a record you declare, and json.encode_with(Config, cfg, ...) writes it back, indented. Nothing is registered anywhere: the library reads the shape of the type while compiling and writes the decoder for it. #[json("host_name")] renames a key, #[json(skip)] leaves a field out, text? accepts null or a missing key, and unknown keys are ignored. A wrong value is an error that carries its path (at $.port); json.parse_value gives a free-form tree when there is no type to read into. See the json library.

web: a server, a template and the client

examples/web.zila
// The web library: a server, an HTML template and the client in one program.
//
//   zila examples/web.zila -o web.exe
//
// The server listens on a free port (0) in the background; the client asks it for a page. An
// `html` block is markup with `{expression}` holes: every value is escaped, so the user name below
// cannot inject markup. `@for`, `@if`, `@else` and `@end` lines make loops and branches.

package main

use web

type Item = record
    name: text
    price: int
end

func page(user: text, items: []Item) -> web.Safe
    return html
        <h1>Hello, {user}</h1>
        <ul>
        @for item in items
            @if item.price > 100
                <li><b>{item.name}</b> costs {item.price} (expensive)</li>
            @else
                <li>{item.name} costs {item.price}</li>
            @end
        @end
        </ul>
    end
end

func main() -> error
    let items = [Item { name: "pen", price: 3 }, Item { name: "<laptop>", price: 900 }]
    var app = web.server(port: 0)
    app.get("/", func(req)
        return web.html_page(200, page(req.query_param("user") ?? "world", items))
    end)
    app.get("/ping", func(req)
        return web.text(200, "pong")
    end)
    let running = try app.start()
    let base = "http://127.0.0.1:{running.port}"

    let pong = try web.get(base + "/ping")
    print("{pong.status} {pong.body}")
    let home = try web.get(base + "/?user=%3Cb%3EAna%3C/b%3E")
    print(home.status)
    print(home.body)
    running.stop()
    return
end

The program starts a server on a free port in the background (app.start() returns at once; app.run() would serve until the program ends), asks it for a page with web.get and prints the reply. The page comes from an html block: markup with {expression} holes and @for, @if, @else, @end lines. Every value is escaped, which is why the user name <b>Ana</b> and the item <laptop> come out as text. The block yields a web.Safe, so one template can be placed inside another without being escaped twice.
