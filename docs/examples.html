Hello world

examples/hello.br
package main

use http, db, json

type Customer = record
    id: i32
    name: text
    email: text?
end

func greet(c: Customer) -> text
    return "Hello, {c.name}!"
end

func main()
    var c = Customer{ id: 1, name: "Ana" }
    print(greet(c))

    for i in 1..5
        if i % 2 == 0
            print("{i} is even")
        end
    end
end

This is the smallest complete Brasa program that still shows most of the language's flavor: a record type with an optional field (email: text?), a free function (greet) that builds a string with interpolation, a record literal (Customer{ id: 1, name: "Ana" }), and a for ... in loop over a range (1..5) with an if inside it. Running it with --run prints Hello, Ana! followed by the even numbers from 1 to 5.

REST API

examples/rest_api.br
package main

use http, db, db.postgres, json

type Customer = record
    id: i32
    name: text
end

func main() -> error
    let conn = try db.open("postgres://localhost/shop")
    defer conn.close()

    var app = http.server(port: 8080)

    app.get("/customers", func(req)
        let rows = try conn.query(Customer, "select id, name from customers")
        return http.json(rows)
    end)

    app.get("/customers/:id", func(req)
        let c = try conn.one(Customer, "select * from customers where id = $1", req.param("id"))
        return http.json(c)
    end)

    app.post("/customers", func(req)
        let c = try req.body(Customer)
        try conn.exec("insert into customers(name) values($1)", c.name)
        return http.created(c)
    end)

    app.run()
end

This example shows error propagation and resource cleanup working together: main is declared -> error (it can fail), so every fallible call starts with try — try db.open(...), the queries, try req.body(Customer) — and any failure short-circuits the function immediately. defer conn.close() guarantees the connection is released whenever main returns, success or failure. Route handlers are passed as inline lambdas (func(req) ... end), and conn.query(Customer, "...") passes the Customer type itself as a value so the database layer knows how to shape the result rows — no reflection or code generation required.

Type system tour

examples/types.br
package main

type Printable = interface
    func to_text(self) -> text
end

type Shape = enum
    Circle(float)
    Rect(float, float)
    Empty
end

type Color = enum
    Red, Green, Blue
end

type Stack<T> = record
    items: []T
end

type Handler = func(Request) -> Response or error
type Middleware = func(Request, next: Handler) -> Response or error
type Table = Map<text, List<i32>>

type User = record
    id: i32
    #[json("full_name")] name: text
    #[json(skip)] password: text
    email: text?
end

func first<T>(items: []T) -> T?
    if items.len == 0
        return nil
    end
    return items[0]
end

func show<T: Printable>(x: T)
    print(x.to_text())
end

func (c: Customer) greet() -> text
    return "Hello, {c.name}"
end

func (ref s: Server) use(m: Middleware)
    s.chain.add(m)
end

func area(s: Shape) -> float
    match s
        Circle(r)  -> return 3.14159 * r * r
        Rect(w, h) -> return w * h
        Empty      -> return 0.0
    end
end

func classify(n: i32) -> text
    match n
        0 -> return "zero"
        1 | 2 | 3 ->
            let s = "small"
            return s
        -1 -> return "minus one"
        _ -> return "many"
    end
end

func fill(ref buf: Buffer, n: i32 = 4)
    buf.n = n
end

func log(msg: text, args: ...any)
    print(msg)
end

func main()
    var b = Buffer{}
    fill(ref b)
    fill(ref b, n: 8)
    let xs = [1, 2, 3]
    log("x", ...xs)
end

This file is a tour of Brasa's type declarations in one place: a structural interface (Printable), an enum with data (Shape) consumed exhaustively by match in area, an enum (Color), a generic record (Stack<T>), function-type aliases (Handler, Middleware), and a record with field attributes controlling JSON (de)serialization (User). It also demonstrates generic functions with an interface bound (show<T: Printable>), a method with a ref receiver ((ref s: Server) use(...)), multi-way match patterns (1 | 2 | 3), a default parameter (n: i32 = 4), and a variadic parameter spread at the call site (log("x", ...xs)).

defer: cleanup that cannot be forgotten

examples/defer.br
// defer: run a call when the function ends, however it ends.
//
//   brasa-parse examples/defer.br -o defer.exe
//
// Deferred calls run last in, first out, on `return`, at the end of the
// function, and when a `try` fails.

package main

type Log = record
    name: text
end

func (l: Log) line(message: text)
    print("[{l.name}] {message}")
end

func process(id: int) -> int or error
    let log = Log{ name: "job {id}" }
    log.line("start")
    defer log.line("finished")      // runs when process() ends, however it ends

    if id < 0
        return error("negative id")
    end
    defer log.line("cleaning up")   // runs first: last in, first out
    return id * 2
end

func main() -> error
    print("result: {try process(21)}")
    match process(-1)
        ok(v)  -> print("result: {v}")
        err(e) -> print("failed: {e.message}")
    end
    return
end

Each defer schedules a call for the moment process ends. They run last in, first out, so "cleaning up" comes before "finished". They run whether the function returns normally, returns early, or fails with try: the failed call for -1 still prints "finished". See the syntax guide for the exact rules.

Maps: look values up by key

examples/maps.br
// Maps: look values up by key.
//
//   brasa-parse examples/maps.br -o maps.exe
//
// A Map<K, V> keeps values under keys (text or integers) and remembers the order
// in which the keys were first added.

package main

func count_words(words: List<text>) -> Map<text, int>
    var counts: Map<text, int> = {}
    for w in words
        counts[w] = counts.get(w, 0) + 1
    end
    return counts
end

func main()
    let words = ["red", "green", "red", "blue", "green", "red"]
    let counts = count_words(words)

    for word, n in counts
        print("{word}: {n}")
    end

    let prices = { "apple": 3, "pear": 5 }
    prices["plum"] = 4
    prices.remove("pear")
    print("{prices.len} prices, apple costs {prices["apple"]}")

    if prices.has("pear")
        print("pears are back")
    else
        print("no pears today")
    end

    print("kiwi costs {prices.get("kiwi", 0)}")
end

count_words builds a Map<text, int> that counts how often each word appears: counts.get(w, 0) is the current count, or 0 for a word not seen yet. The for word, n in counts loop visits the entries in the order the words were first added (red, green, blue), so the output never changes between runs. The second half shows a literal ({ "apple": 3, "pear": 5 }), adding and removing keys, len and has. See the syntax guide for the exact rules.

Text and lists: count the words

examples/words.br
// Text and list helpers: count the words in a sentence.
//
//   brasa-parse examples/words.br -o words.exe
//
// `use text` and `use list` are libraries written in Brasa. Their functions are called with
// the module name in front, like text.split(...) and list.sort_by(...).

package main

use text
use list

type Entry = record
    word: text
    count: int
end

func main()
    let sentence = "The quick brown fox jumps over the lazy dog. The dog barks; the fox runs."

    // lower-case it, turn punctuation into spaces and cut it into words
    var cleaned = text.lower(sentence)
    cleaned = text.replace(cleaned, ".", " ")
    cleaned = text.replace(cleaned, ";", " ")
    let words = list.filter(text.split(cleaned, " "), func(w) w.len > 0 end)

    var counts: Map<text, int> = {}
    for w in words
        counts[w] = counts.get(w, 0) + 1
    end

    let entries: []Entry = []
    for w, n in counts
        entries.add(Entry { word: w, count: n })
    end

    // most frequent first, ties in alphabetical order
    let ranked = list.sort_by(entries, func(a, b) a.count > b.count or (a.count == b.count and text.compare(a.word, b.word) < 0) end)
    for i in 0..3
        let e = ranked[i]
        print("{text.pad_right(e.word, 6, " ")} {e.count}")
    end

    let total = list.reduce(list.map(ranked, func(e) e.count end), 0, func(sum, n) sum + n end)
    print("{words.len} words, {ranked.len} different, {total} counted")

    let first = text.join(list.map(list.slice(ranked, 0, 3), func(e) text.upper(e.word) end), ", ")
    print("top three: {first}")

    // numbers typed as text
    let n = text.parse_int("42") ?? 0
    let bad = text.parse_int("4x2") ?? 0 - 1
    let price = text.parse_float("19.99") ?? 0.0
    print("{n} {bad} {price * 2.0}")
end

This program cuts a sentence into words with text.split, drops the empty pieces with list.filter, counts them in a map, and ranks them with list.sort_by: the function you pass says that a higher count goes first and ties go in alphabetical order (text.compare). The last lines show text.parse_int and text.parse_float, which give a default with ?? when the text is not a number. The Library page lists every function.

Generics: one function, many types

examples/generics.br
// Generics and interfaces: write code once, use it with many types.
//
//   brasa-parse examples/generics.br -o generics.exe
//
// <T> is a placeholder for a type. The compiler works out T from the arguments and
// makes one copy of the code for each type you use. An interface lists methods, and a
// bound (<T: Printable>) says "any type that has these methods".

package main

type Printable = interface
    func to_text(self) -> text
end

type Point = record
    x: int
    y: int
end

type Dog = record
    name: text
end

func (p: Point) to_text() -> text
    return "({p.x}, {p.y})"
end

func (d: Dog) to_text() -> text
    return "dog {d.name}"
end

func (n: int) to_text() -> text
    return "the number {n}"
end

type Stack<T> = record
    items: []T
end

func (s: Stack<T>) push(x: T)
    s.items.add(x)
end

func (s: Stack<T>) pop() -> T
    return s.items.take()
end

func first<T>(items: []T) -> T
    return items[0]
end

func show<T: Printable>(x: T)
    print("showing {x.to_text()}")
end

func main()
    let numbers: Stack<int> = Stack { items: [] }
    numbers.push(3)
    numbers.push(4)
    print("popped {numbers.pop()}")

    let names: Stack<text> = Stack { items: ["ada"] }
    names.push("alan")
    print("popped {names.pop()}")

    print("first of two lists: {first([10, 20])} and {first(["x", "y"])}")

    show(Point { x: 1, y: 2 })
    show(Dog { name: "rex" })
    show(42)
end

Stack<T> and first<T> are written once and used with int and text; the compiler works out T from the arguments and the declared type of the variable. Printable lists one method, and show<T: Printable> accepts anything that has it: a record with a to_text method, or even int, because func (n: int) to_text() gives it one. There is no implements keyword. See the syntax guide for the exact rules.

Floats: fractional numbers

examples/floats.br
// Floating point: fractional numbers.
//
//   brasa-parse examples/floats.br -o floats.exe
//
// `float` is a 64-bit number with a fractional part. It never mixes with `int`
// by accident: convert with float(n) and int(x).

package main

func average(values: List<float>) -> float
    var total = 0.0
    for v in values
        total += v
    end
    return total / float(values.len)
end

func compound(principal: float, rate: float, years: int) -> float
    var amount = principal
    for year in 0..years
        amount = amount * (1.0 + rate)
    end
    return amount
end

func main()
    let temps = [21.5, 19.0, 23.25, 18.75]
    print("average {average(temps)}")

    // 0.1 has no exact binary form, and Brasa shows the shortest text that reads back exactly
    print("0.1 + 0.2 = {0.1 + 0.2}")
    print("1 / 3 = {1.0 / 3.0}")

    print("savings {compound(1000.0, 0.05, 10)}")

    let whole = 7
    print("half of {whole} is {float(whole) / 2.0}")
    print("cut off: {int(3.99)} and {int(-3.99)}")

    // division by zero is not an error for floats
    let zero = 0.0
    print("{1.0 / zero} {-1.0 / zero} {zero / zero}")

    // NaN is never equal to anything, itself included
    let nan = zero / zero
    print("{nan == nan} {nan != nan} {1.5 < 2.5}")
end

average adds up a list of float values and divides by float(values.len): a float never mixes with an int by accident, so the count is converted on purpose. Printing gives the shortest text that reads back as exactly the same number, which is why 0.1 + 0.2 shows 0.30000000000000004. Dividing by zero is not an error for floats (it gives inf, -inf or NaN), and NaN is not equal to anything, itself included. See the syntax guide for the exact rules.

Native windows

examples/ui.br
package main

use ui

func main()
    var win = ui.Window("Customers", 640, 480)

    var name = ui.TextBox()
    var list = ui.ListView()
    var add  = ui.Button("Add")

    add.onClick(func()
        list.add(name.text)
        name.text = ""
    end)

    win.layout(ui.vstack(name, add, list))
    win.show()
    ui.run()
end

The ui library is written in Brasa on top of the Win32 API. ui.Window, ui.TextBox, ui.Button and ui.ListView are real native controls; vstack lays them out; the lambda passed to onClick runs when the button is pressed (or Enter is pressed in the text box). Build it with brasa-parse examples/ui.br -o ui.exe.

Async: waiting, files and TCP

examples/async.br
// async/await: waiting, files and TCP in one program.
//
//   brasa-parse examples/async.br -o async.exe
//
// An `async func` starts running when it is called and returns a Future.
// `await` waits for a future without blocking the other tasks.

package main

use fs
use net

async func fetch_title(id: int) -> text
    await sleep(100)
    return "title {id}"
end

async func echo_server(listener: net.Listener) -> error
    let conn = try await listener.accept()
    let message = try await conn.read(64)
    try await conn.write("echo: {message}")
    await conn.close()
    return
end

func main() -> error
    // three 100 ms waits overlap: all of it takes about 100 ms, not 300
    let titles = await all([fetch_title(1), fetch_title(2), fetch_title(3)])
    print("{titles[0]}, {titles[1]}, {titles[2]}")

    // files
    try await fs.write("brasa-example.txt", "written by an async function")
    print(try await fs.read("brasa-example.txt"))

    // a TCP server and a client in the same program
    let listener = try await net.listen(0)
    let server = echo_server(listener)
    let client = try await net.connect("127.0.0.1", listener.port)
    try await client.write("hello")
    print(try await client.read(64))
    try await server
    return
end

Three one-hundred-millisecond waits overlap, so await all(...) takes about 100 ms. The program then writes and reads a file and runs a TCP server and client in the same process; try await turns any failure into an early return. See Async.

FFI & low-level

The whole of examples/ffi.br. See brasa-ffi.md for the full FFI and inline-assembly specification.

examples/ffi.br
package main

use c

// ---- calling C ----

#[link("m")]
extern "C" func sqrt(x: f64) -> f64

extern "C" func printf(fmt: *c.char, ...) -> c.int

extern "C" func qsort(base: *c.void, n: c.size_t, size: c.size_t,
    cmp: extern "C" func(*c.void, *c.void) -> c.int)

#[link_src("vendor/crc.c")]
extern "C" func crc32(data: *u8, len: usize) -> u32

#[stdcall_example]
extern "stdcall" func MessageBoxA(hwnd: usize, text: *c.char, caption: *c.char, kind: u32) -> c.int

#[c_layout]
type Point = record
    x: f64
    y: f64
end

// ---- being called from C ----

#[export("brasa_add")]
func add(a: i32, b: i32) -> i32
    return a + b
end

// ---- inline assembly ----

#[cfg("x86_64")]
func rdtsc() -> u64
    var lo: u32 = 0
    var hi: u32 = 0
    unsafe
        asm
            rdtsc
            out eax = lo
            out edx = hi
        end
    end
    return u64(hi) << 32 bitor u64(lo)
end

#[cfg("x86_64")]
func add_asm(a: u64, b: u64) -> u64
    var r: u64 = a
    unsafe
        asm
            add @r, @b
            inout r = r
            in b = b
            clobber cc
        end
    end
    return r
end

#[naked]
#[cfg("x86_64")]
func halt()
    asm
        cli
        hlt
    end
end

func main()
    unsafe
        printf(c.cstr("sqrt(2) = %f\n"), sqrt(2.0))
    end
    print("tsc: {rdtsc()}")
end

This program exercises Brasa on both sides of the C boundary. extern "C" func sqrt(...) and printf declare foreign functions pulled in via #[link("m")] (link against libm); printf's trailing ... marks it as C-variadic. qsort's cmp parameter shows an extern "C" func(...) type used as a C-compatible function-pointer callback. #[c_layout] on Point pins field order and alignment to match a C struct exactly, so it can be passed across the FFI boundary safely. #[export("brasa_add")] does the reverse — it exposes a Brasa function to C callers under a fixed, unmangled symbol name. rdtsc reads the CPU timestamp counter with a short inline asm block, using out eax = lo / out edx = hi to bind two hardware registers to Brasa variables. Every one of these operations — calling extern functions and running asm — is wrapped in unsafe, since the compiler can't verify what happens on the other side of the boundary. See the FFI section of the syntax guide and the project's brasa-ffi.md for the full specification.
